Danh mục Marketplace

File marketplace.json liệt kê plugin và nơi tải chúng.

File liên quan
.claude/plugins/MARKETPLACES.md
.claude/plugins/my-marketplace/.claude-plugin/marketplace.json
json
{
  // .claude-plugin/marketplace.json — Danh mục Marketplace
  //
  // File này định nghĩa một plugin marketplace: danh mục liệt kê
  // plugin và cho Claude Code biết nơi tải chúng. Host nó trong
  // một git repo và chia sẻ cho nhóm hoặc cộng đồng của bạn.
  //
  // Người dùng thêm marketplace của bạn:
  //   /plugin marketplace add owner/repo
  //
  // Rồi cài từng plugin:
  //   /plugin install review-plugin@my-marketplace

  // ── Danh tính marketplace ──────────────────────────────────────

  // Tên công khai (kebab-case). Người dùng thấy tên này khi cài:
  //   /plugin install <plugin>@my-marketplace
  "name": "my-marketplace",

  // Ai duy trì marketplace này
  "owner": {
    "name": "Nhóm của bạn",
    "email": "team@example.com"
  },

  // ── Metadata tuỳ chọn ──────────────────────────────────────────
  "metadata": {
    "description": "Marketplace khởi đầu minh hoạ định dạng danh mục",
    "version": "1.0.0",

    // Thư mục gốc cho nguồn plugin tương đối.
    // Khi đặt giá trị này, "source": "review-plugin" phân giải thành "./plugins/review-plugin"
    "pluginRoot": "./plugins"
  },

  // ── Danh mục plugin ────────────────────────────────────────────
  //
  // Mỗi mục tối thiểu cần name và source.
  // Bạn cũng có thể thêm bất kỳ trường nào trong schema plugin.json.

  "plugins": [

    // ── Nguồn đường dẫn tương đối ────────────────────────────────
    // Plugin nằm cùng repo với marketplace.
    // Lựa chọn đơn giản nhất cho marketplace tự chứa.
    {
      "name": "review-plugin",
      "source": "./plugins/review-plugin",
      "description": "Thêm skill /review để review code nhanh",
      "version": "1.0.0"
    },

    // ── Nguồn GitHub ─────────────────────────────────────────────
    // Plugin nằm trong một repo GitHub riêng.
    // Hỗ trợ ghim ref (nhánh/tag) và sha (commit chính xác).
    {
      "name": "deploy-tools",
      "source": {
        "source": "github",
        "repo": "your-org/deploy-plugin",
        "ref": "v2.0.0"
      },
      "description": "Tự động hoá deploy và quản lý môi trường",
      "category": "devops",
      "tags": ["deploy", "ci", "automation"]
    },

    // ── Nguồn npm ────────────────────────────────────────────────
    // Plugin phân phối dưới dạng package npm.
    // Hỗ trợ dải phiên bản và registry tuỳ chỉnh.
    {
      "name": "lint-plugin",
      "source": {
        "source": "npm",
        "package": "@your-org/claude-lint-plugin",
        "version": "^1.0.0"
      },
      "description": "Hook tự động lint cho JavaScript và TypeScript"
    }

    // Các loại nguồn khác:
    //
    //   Git URL:
    //   { "source": "url", "url": "https://gitlab.com/team/plugin.git" }
    //
    //   Git subdirectory (sparse clone cho monorepo):
    //   { "source": "git-subdir", "url": "https://...", "path": "tools/plugin" }
    //
    //   pip:
    //   { "source": "pip", "package": "my-plugin" }
  ]
}
