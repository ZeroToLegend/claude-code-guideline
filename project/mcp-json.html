MCP

Cấu hình MCP

Kết nối Claude với công cụ, API và dịch vụ bên ngoài qua Model Context Protocol.

.mcp.json
json
{
  // .mcp.json — Cấu hình MCP phạm vi dự án
  //
  // MCP (Model Context Protocol) kết nối Claude với công cụ, API
  // và nguồn dữ liệu bên ngoài. Mỗi "server" cung cấp một bộ công cụ
  // mà Claude có thể khám phá và gọi trong phiên làm việc.
  //
  // File này nằm ở gốc dự án. Hãy commit nó vào git để mọi thành
  // viên trong nhóm có cùng cấu hình công cụ.
  //
  // Hỗ trợ ba loại server:
  //   "http"    — Server HTTP từ xa (khuyên dùng cho công cụ remote)
  //   "command" — Tiến trình stdio cục bộ (khởi chạy mỗi phiên)
  //   "url"     — Endpoint SSE (lỗi thời, hãy dùng http thay thế)
  //
  // Cài server bằng CLI thay vì sửa tay file này:
  //   claude mcp add my-server -s project -- npx some-mcp-package
  //
  // Phạm vi (scope) quyết định ai thấy từng server:
  //   -s local    — Chỉ bạn, máy này (~/.claude.json)
  //   -s project  — Mọi người trong dự án (.mcp.json, được commit)
  //   -s user     — Chỉ bạn, mọi dự án (~/.claude/settings.json)
  //
  // Quản lý server trong phiên bằng lệnh slash /mcp.

  "mcpServers": {

    // ── HTTP server (khuyên dùng cho dịch vụ từ xa) ────────────
    // Kết nối tới một endpoint HTTP từ xa. Hỗ trợ OAuth nếu server
    // yêu cầu xác thực. Claude tự xử lý luồng đăng nhập.
    "my-remote-service": {
      "type": "http",
      "url": "https://my-service.example.com/mcp"
    },

    // ── stdio server (cho công cụ cục bộ) ──────────────────────
    // Khởi chạy một tiến trình cục bộ. Claude giao tiếp qua stdin/stdout.
    // Phần lớn gói MCP trên npm/pip dùng kiểu truyền này.
    "my-local-tool": {
      "command": "npx",
      "args": ["-y", "@example/mcp-server-tool"],
      "env": {
        // Biến môi trường hỗ trợ cú pháp mở rộng:
        //   ${env:VAR}            — đọc từ môi trường shell của bạn
        //   ${env:VAR:-default}   — dùng "default" nếu chưa đặt
        "API_KEY": "${env:MY_TOOL_API_KEY}",
        "DATABASE_URL": "${env:DATABASE_URL:-postgresql://localhost:5432/mydb}"
      },
      // Tuỳ chọn: đặt thư mục làm việc cho tiến trình được khởi chạy
      "cwd": "/path/to/working/directory"
    },

    // ── Ví dụ thực tế ──────────────────────────────────────────
    // Một số server cộng đồng phổ biến bạn có thể thêm:
    //
    //   claude mcp add github -s project -- npx -y @modelcontextprotocol/server-github
    //   claude mcp add postgres -s local -- npx -y @modelcontextprotocol/server-postgres
    //   claude mcp add sentry -s project -- npx -y @sentry/mcp-server-sentry
    //
    // Xem danh sách server: https://github.com/modelcontextprotocol/servers

    // ── Tool Search ────────────────────────────────────────────
    // Khi MCP server cung cấp quá nhiều công cụ, Claude dùng Tool Search
    // để tìm đúng công cụ cần thay vì nạp tất cả. Việc này tự động xảy ra
    // với server có từ 30 công cụ trở lên. Để điều chỉnh:
    //
    //   "toolSearch": {
    //     "enabled": "auto",        — "auto" (mặc định), true, hoặc false
    //     "threshold": 30           — số công cụ để kích hoạt tìm kiếm
    //   }

    // ── MCP Resources ──────────────────────────────────────────
    // Một số server cung cấp dữ liệu dưới dạng resource (tài liệu, schema...).
    // Tham chiếu trong prompt bằng @server-name:resource-id
    // để kéo dữ liệu đó vào bối cảnh.

    // ── MCP Prompts ────────────────────────────────────────────
    // Server có thể cung cấp prompt mẫu tái sử dụng, xuất hiện dưới
    // dạng lệnh slash. Chạy /mcp trong phiên để xem các prompt có sẵn.
  }
}
